/* * O dialog é centralizado no momento em que abre, mas a imagem * ainda não terminou de baixar nessa hora. Quando ela carrega, a * altura do dialog muda e ele fica deslocado. Por isso * recentralizamos: ao abrir, quando cada imagem termina de * carregar e ao redimensionar a janela. */ function dlgImagemCentralizar() { var widget = PF('dlgVisualizarImagem'); if (!widget) { return; } widget.initPosition(); // Recentraliza assim que a imagem terminar de carregar widget.jq.find('img').each(function () { var img = this; if (img.complete) { widget.initPosition(); } else { img.onload = function () { widget.initPosition(); }; } }); // Rede de segurança para fontes/estilos que assentam depois setTimeout(function () { widget.initPosition(); }, 250); } // Mantém centralizado ao redimensionar a janela if (window.jQuery) { jQuery(window).on('resize', function () { var widget = PF('dlgVisualizarImagem'); if (widget && widget.isVisible()) { widget.initPosition(); } }); }
